**MobileWeekStrip.jsx:**

Horizontal strip of 7 day tabs with prev/next week arrows. Day tab shows day initial letter ("M", "T", "O", "T", "F", "L", "S" for Norwegian Mon-Sun) and date number. Selected day tab gets primary fill.

```jsx
import { ChevronLeft, ChevronRight } from "lucide-react";
import { weekDays, getIsoWeekNumber } from "../../lib/dateUtils";

const DAY_INITIALS = ["M", "T", "O", "T", "F", "L", "S"]; // Norwegian: Mandag, Tirsdag, Onsdag, Torsdag, Fredag, Lordag, Sondag

export function MobileWeekStrip({ weekStart, selectedDay, onSelectDay, onPrev, onNext }) {
  const days = weekDays(weekStart);
  const weekNum = getIsoWeekNumber(weekStart);

  return (
    <div>
      {/* Week label + nav arrows */}
      <div className="flex items-center justify-between px-4 py-2">
        <button type="button" onClick={onPrev} aria-label="Previous week">
          <ChevronLeft size={20} style={{ color: "var(--pa-on-surface)" }} />
        </button>
        <span
          style={{
            fontSize: "var(--pa-text-body-md)",
            fontFamily: "var(--pa-font-display)",
            fontWeight: 700,
            color: "var(--pa-on-surface)",
          }}
        >
          Uke {weekNum}
        </span>
        <button type="button" onClick={onNext} aria-label="Next week">
          <ChevronRight size={20} style={{ color: "var(--pa-on-surface)" }} />
        </button>
      </div>

      {/* Day tabs */}
      <div className="flex justify-between px-4" style={{ gap: "var(--pa-space-1)" }}>
        {days.map((iso, i) => {
          const isSelected = iso === selectedDay;
          const dayNum = new Date(`${iso}T00:00:00Z`).getUTCDate();
          return (
            <button
              key={iso}
              type="button"
              onClick={() => onSelectDay(iso)}
              className="flex flex-col items-center justify-center flex-1"
              style={{
                minHeight: "44px",
                borderRadius: "var(--pa-radius-card)",
                background: isSelected ? "var(--pa-primary)" : "var(--pa-surface-container-low)",
                color: isSelected ? "var(--pa-on-primary)" : "var(--pa-on-surface)",
                transition: "background 150ms ease, color 150ms ease",
              }}
            >
              <span
                style={{
                  fontSize: "var(--pa-text-label-sm)",
                  fontFamily: "var(--pa-font-body)",
                  fontWeight: 700,
                }}
              >
                {DAY_INITIALS[i]}
              </span>
              <span
                style={{
                  fontSize: "var(--pa-text-label-sm)",
                  fontFamily: "var(--pa-font-body)",
                  fontWeight: 400,
                }}
              >
                {dayNum}
              </span>
            </button>
          );
        })}
      </div>
    </div>
  );
}
```

**MobileDayPanel.jsx:**

Shows all workout cards for the selected day, reusing the same card primitives as desktop. Empty state shows subtle prompt text.

```jsx
import { normalizeWorkoutType } from "../../domain/workoutTypes";
import { WorkoutCard } from "./WorkoutCard";
import { RaceCard } from "./RaceCard";
import { RestCard } from "./RestCard";
import { ConstraintMarker } from "./ConstraintMarker";
import { formatDayLabel } from "../../lib/dateUtils";

export function MobileDayPanel({ isoDate, entries = [] }) {
  const constraintEntry = entries.find(
    (e) => normalizeWorkoutType(e.workout_type) === "CONSTRAINT" ||
           (e.workout_type && e.workout_type.toLowerCase() === "constraint")
  );
  const workoutEntries = entries.filter(
    (e) => normalizeWorkoutType(e.workout_type) !== "CONSTRAINT" &&
           !(e.workout_type && e.workout_type.toLowerCase() === "constraint")
  );

  function renderEntry(entry) {
    const typeKey = normalizeWorkoutType(entry.workout_type);
    if (typeKey === "REST") return <RestCard key={entry.id} />;
    if (typeKey === "RACE_EVENT") return <RaceCard key={entry.id} entry={entry} />;
    return <WorkoutCard key={entry.id} entry={entry} />;
  }

  return (
    <div className="px-4 py-3">
      <div
        style={{
          fontSize: "var(--pa-text-label-md)",
          fontFamily: "var(--pa-font-body)",
          fontWeight: 700,
          color: "var(--pa-on-surface-variant)",
          textTransform: "uppercase",
          marginBottom: "var(--pa-space-2)",
        }}
      >
        {formatDayLabel(isoDate)}
      </div>
      {workoutEntries.length === 0 && !constraintEntry ? (
        <div
          style={{
            fontSize: "var(--pa-text-label-sm)",
            fontFamily: "var(--pa-font-body)",
            color: "var(--pa-on-surface-variant)",
            opacity: 0.5,
            padding: "var(--pa-space-4) 0",
          }}
        >
          Ingen okter planlagt
        </div>
      ) : (
        <div className="flex flex-col" style={{ gap: "var(--pa-space-2)" }}>
          {workoutEntries.map(renderEntry)}
          {constraintEntry && <ConstraintMarker reason={constraintEntry.description} />}
        </div>
      )}
    </div>
  );
}
```

**MobileLoadSummary.jsx:**

Single-line compact load display: "{N} km \u00b7 {time}". No stacked bar per CONTEXT.md mobile decision.

```jsx
import { useMemo } from "react";
import { computeWeeklyLoadStats } from "../../domain/compute";

export function MobileLoadSummary({ entries = [], targetKm = null }) {
  const stats = useMemo(
    () => computeWeeklyLoadStats(entries, targetKm),
    [entries, targetKm]
  );

  const { totalKm, totalHours } = stats;
  const isEmpty = totalKm === 0 && totalHours === 0;

  const hours = Math.floor(totalHours);
  const mins = Math.round((totalHours - hours) * 60);
  const timeLabel = hours > 0
    ? `${hours}h${mins > 0 ? ` ${mins}m` : ""}`
    : `${mins}m`;

  const kmLabel = totalKm >= 10
    ? `${Math.round(totalKm)} km`
    : `${totalKm.toFixed(1)} km`;

  return (
    <div
      className="px-4 py-2 text-center"
      style={{
        fontSize: "var(--pa-text-label-md)",
        fontFamily: "var(--pa-font-body)",
        fontWeight: 400,
        color: "var(--pa-on-surface-variant)",
      }}
    >
      {isEmpty ? "\u2014" : `${kmLabel} \u00b7 ${timeLabel}`}
    </div>
  );
}
```

**MobilePlannerPager.jsx:**

Root mobile component with swipe navigation and day selection. Loads entries via `loadEntriesForRange` for the visible week only (re-fetches on week change). Uses touch swipe detection with horizontal-only guard (Pitfall 4 avoidance).

```jsx
import { useEffect, useMemo, useRef, useState } from "react";
import { currentMondayIso, isoDateOffset, todayIso, getWeekIntent } from "../../lib/dateUtils";
import { MobileWeekStrip } from "./MobileWeekStrip";
import { MobileDayPanel } from "./MobileDayPanel";
import { MobileLoadSummary } from "./MobileLoadSummary";

export function MobilePlannerPager({ entries = [], loading, loadEntriesForRange, planId, blocks = [] }) {
  const [weekStart, setWeekStart] = useState(() => currentMondayIso());
  const [selectedDay, setSelectedDay] = useState(() => todayIso());

  // Load entries for the visible week
  useEffect(() => {
    if (!planId || !weekStart) return;
    const endIso = isoDateOffset(weekStart, 6);
    loadEntriesForRange(planId, weekStart, endIso);
  }, [planId, weekStart, loadEntriesForRange]);

  const weekEntries = useMemo(() => {
    const we = isoDateOffset(weekStart, 6);
    return entries.filter((e) => e.workout_date >= weekStart && e.workout_date <= we);
  }, [entries, weekStart]);

  const dayEntries = useMemo(
    () => entries.filter((e) => e.workout_date === selectedDay),
    [entries, selectedDay]
  );

  const intent = useMemo(
    () => getWeekIntent(blocks, planId, weekStart),
    [blocks, planId, weekStart]
  );

  const goNext = () => {
    const next = isoDateOffset(weekStart, 7);
    setWeekStart(next);
    setSelectedDay(next); // Select Monday of new week
  };
  const goPrev = () => {
    const prev = isoDateOffset(weekStart, -7);
    setWeekStart(prev);
    setSelectedDay(prev);
  };

  // Touch swipe detection — horizontal-only guard (Pitfall 4)
  const touchStart = useRef(null);
  const onTouchStart = (e) => {
    touchStart.current = { x: e.touches[0].clientX, y: e.touches[0].clientY };
  };
  const onTouchEnd = (e) => {
    if (!touchStart.current) return;
    const deltaX = e.changedTouches[0].clientX - touchStart.current.x;
    const deltaY = e.changedTouches[0].clientY - touchStart.current.y;
    touchStart.current = null;
    // Only trigger if horizontal movement dominates vertical AND exceeds threshold
    if (Math.abs(deltaX) > Math.abs(deltaY) && Math.abs(deltaX) > 50) {
      deltaX < 0 ? goNext() : goPrev();
    }
  };

  return (
    <div
      onTouchStart={onTouchStart}
      onTouchEnd={onTouchEnd}
      style={{ background: "var(--pa-surface)" }}
    >
      <MobileWeekStrip
        weekStart={weekStart}
        selectedDay={selectedDay}
        onSelectDay={setSelectedDay}
        onPrev={goPrev}
        onNext={goNext}
      />
      <MobileLoadSummary entries={weekEntries} targetKm={intent?.targetMileageKm ?? null} />
      <MobileDayPanel isoDate={selectedDay} entries={dayEntries} />
    </div>
  );
}
```

**DayCell.jsx:**
```jsx
import { normalizeWorkoutType } from "../../domain/workoutTypes";
import { formatDayLabel, todayIso } from "../../lib/dateUtils";
import { WorkoutCard } from "./WorkoutCard";
import { RaceCard } from "./RaceCard";
import { RestCard } from "./RestCard";
import { ConstraintMarker } from "./ConstraintMarker";
import { AddAffordance } from "./AddAffordance";

export function DayCell({ isoDate, entries = [] }) {
  const isToday = isoDate === todayIso();
  const constraintEntry = entries.find(
    (e) => normalizeWorkoutType(e.workout_type) === "CONSTRAINT" ||
           (e.workout_type && e.workout_type.toLowerCase() === "constraint")
  );
  const workoutEntries = entries.filter(
    (e) => normalizeWorkoutType(e.workout_type) !== "CONSTRAINT" &&
           !(e.workout_type && e.workout_type.toLowerCase() === "constraint")
  );

  // Render a card based on the normalized workout type
  function renderEntry(entry) {
    const typeKey = normalizeWorkoutType(entry.workout_type);
    if (typeKey === "REST") return <RestCard key={entry.id} />;
    if (typeKey === "RACE_EVENT") return <RaceCard key={entry.id} entry={entry} />;
    return <WorkoutCard key={entry.id} entry={entry} />;
  }

  return (
    <div
      className="rounded-xl relative flex flex-col"
      style={{
        background: "var(--pa-surface-container-lowest)",
        boxShadow: "var(--pa-shadow-card)",
        borderRadius: "var(--pa-radius-card)",
        padding: "var(--pa-space-3)",
        minHeight: "140px",
        ...(isToday ? { boxShadow: "var(--pa-shadow-card), 0 0 0 2px var(--pa-outline)" } : {}),
      }}
    >
      {/* Date label */}
      <div
        style={{
          fontSize: "var(--pa-text-label-sm)",
          fontFamily: "var(--pa-font-body)",
          fontWeight: 700,
          color: "var(--pa-on-surface-variant)",
          textTransform: "uppercase",
          marginBottom: "var(--pa-space-2)",
        }}
      >
        {formatDayLabel(isoDate)}
      </div>

      {/* Workout cards — stacked with 4px gap */}
      <div className="flex flex-col" style={{ gap: "var(--pa-space-1)" }}>
        {workoutEntries.map(renderEntry)}
      </div>

      {/* Constraint marker — supporting context below workouts */}
      {constraintEntry && <ConstraintMarker reason={constraintEntry.description} />}

      {/* Add affordance — only in truly empty cells */}
      {workoutEntries.length === 0 && !constraintEntry && <AddAffordance />}
    </div>
  );
}
```

Note on constraint detection: Since CNST-01 (custom reason DB column) is deferred, constraints are identified by `workout_type === "Constraint"` in entries. The `normalizeWorkoutType` function falls through to `inferWorkoutTypeFromText` which does not recognize "constraint" — so also check the raw string. If no constraint entries exist in the data, constraint cells simply won't render the marker. This is consistent with the RESEARCH.md open question resolution.

For the constraint-only cell (no workout + constraint present), add dashed border per UI-SPEC No-Line Rule exception:

**FourWeekGrid.jsx:**
```jsx
import { useEffect, useMemo } from "react";
import { currentMondayIso, isoDateOffset, getWeekIntent } from "../../lib/dateUtils";
import { WeekRow } from "./WeekRow";

export function FourWeekGrid({ entries = [], loading, loadEntriesForRange, planId, blocks = [] }) {
  const startMonday = useMemo(() => currentMondayIso(), []);

  // Load all 28 days in ONE call — not 4 separate calls (RESEARCH.md anti-pattern)
  useEffect(() => {
    if (!planId || !startMonday) return;
    const endIso = isoDateOffset(startMonday, 27); // inclusive Sunday of week 4
    loadEntriesForRange(planId, startMonday, endIso);
  }, [planId, startMonday, loadEntriesForRange]);

  // Split entries into 4 per-week buckets
  const weeks = useMemo(() => {
    return Array.from({ length: 4 }, (_, i) => {
      const ws = isoDateOffset(startMonday, i * 7);
      const we = isoDateOffset(ws, 6);
      const weekEntries = entries.filter(
        (e) => e.workout_date >= ws && e.workout_date <= we
      );
      const intent = getWeekIntent(blocks, planId, ws);
      return {
        weekStart: ws,
        entries: weekEntries,
        targetKm: intent?.targetMileageKm ?? null,
      };
    });
  }, [entries, startMonday, blocks, planId]);

  if (loading) {
    return (
      <div className="flex flex-col" style={{ gap: "var(--pa-space-6)" }}>
        {[0, 1, 2, 3].map((i) => (
          <div
            key={i}
            className="grid grid-cols-12"
            style={{ gap: "var(--pa-space-4)" }}
          >
            <div className="col-span-11 grid grid-cols-7" style={{ gap: "var(--pa-space-4)" }}>
              {[0, 1, 2, 3, 4, 5, 6].map((j) => (
                <div
                  key={j}
                  className="animate-pulse"
                  style={{
                    background: "var(--pa-surface-container-low)",
                    borderRadius: "var(--pa-radius-card)",
                    minHeight: "140px",
                  }}
                />
              ))}
            </div>
            <div
              className="col-span-1 animate-pulse"
              style={{
                background: "var(--pa-surface-container-low)",
                borderRadius: "var(--pa-radius-card)",
              }}
            />
          </div>
        ))}
      </div>
    );
  }

  return (
    <div className="flex flex-col" style={{ gap: "var(--pa-space-6)" }}>
      {weeks.map(({ weekStart, entries: weekEntries, targetKm }) => (
        <WeekRow
          key={weekStart}
          weekStart={weekStart}
          entries={weekEntries}
          targetKm={targetKm}
        />
      ))}
    </div>
  );
}
```

- Error state: the `loading`/`error` is handled by the parent WeeklyPlanPage via the existing error display pattern
